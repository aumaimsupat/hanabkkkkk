3

คัดลอกโค้ดสคริปต์ด้านล่างนี้ไปวาง

ลบโค้ดเริ่มต้นทั้งหมดในไฟล์ `Code.gs` และลอกโค้ดพรีเมียมตัวนี้ไปวางแทนที่: 

/* โค้ดสำหรับระบบสั่งซื้อสินค้า Hana BKK (สิทธิ์เป็นสาธารณะ) */
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var timestamp = new Date();
    
    // วนลูปสินค้าแต่ละรายการในหนึ่งออเดอร์เพื่อแยกหมวดหมู่ตาราง
    data.items.forEach(function(item) {
      var sheetName = item.category || "อื่นๆ";
      var sheet = ss.getSheetByName(sheetName);
      
      // หากยังไม่มีแท็บนี้ ระบบจะสร้างแท็บใหม่และใส่หัวคอลัมน์อัตโนมัติ
      if (!sheet) {
        sheet = ss.insertSheet(sheetName);
        sheet.appendRow([
          "วัน-เวลาที่สั่ง (Timestamp)", 
          "ชื่อผู้สั่ง", 
          "แผนก", 
          "หมวดหมู่หลัก", 
          "รายการสินค้า", 
          "จำนวน", 
          "หน่วยเรียก", 
          "หมายเหตุ"
        ]);
        // ตกแต่งหัวข้อหัวตาราง
        sheet.getRange(1, 1, 1, 8)
          .setFontWeight("bold")
          .setBackgroundColor("#F5EBE0")
          .setFontColor("#4A3429");
      }
      
      // บันทึกข้อมูลบรรทัดใหม่
      sheet.appendRow([
        timestamp,
        data.requesterName,
        data.department,
        item.category,
        item.itemName,
        item.quantity,
        item.unit,
        item.remarks || ""
      ]);
    });
    
    return ContentService.createTextOutput(JSON.stringify({
      "status": "success", 
      "message": "บันทึกเรียบร้อยลงตารางแบบแยกหมวดหมู่"
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch(error) {
    return ContentService.createTextOutput(JSON.stringify({
      "status": "error", 
      "message": error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

ทำการติดตั้งใช้งานเว็บแอป (Deploy Web App)

  • กดคลิกปุ่มสีน้ำเงินขวาบน การตั้งค่าใช้งานจริง (Deploy) > การจัดการการใช้งานจริงใหม่ (New deployment)
  • เลือกประเภทเป็น เว็บแอป (Web App) (รูปไอคอนลูกโลก)
  • กรอกรายละเอียดคำอธิบาย: "Hana Ordering System"
  • ช่องผู้ดำเนินการติดตั้ง ให้เลือกเป็น: ฉัน (บัญชีของคุณเอง)
  • ช่องผู้มีสิทธิ์เข้าถึงสำคัญมาก ต้องเลือกเป็น: ทุกคน (Anyone) (เพื่ออนุญาตให้ฟอร์มนี้ส่งยอดเข้าชีทได้)
  • คลิก การใช้งานจริง (Deploy) และอนุมัติสิทธิ์ (Authorize access) สำหรับบัญชีของคุณ
  • คัดลอก URL เว็บแอป (Web App URL) ที่ขึ้นตอนสุดท้าย นำมาวางในกล่องตั้งค่าทางซ้ายมือได้ทันที!
